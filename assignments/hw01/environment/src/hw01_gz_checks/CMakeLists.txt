cmake_minimum_required(VERSION 3.16)
project(hw01_gz_checks)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(hw01_observer SHARED src/observer.cpp)

target_link_libraries(hw01_observer PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS hw01_observer LIBRARY DESTINATION lib)
